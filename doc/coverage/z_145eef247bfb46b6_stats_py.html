26def fixed_extra_waste(sizes: Sequence[int]) -> int: 

27 """Mehrverbrauch W = n * max(s) - sum(s) von ``fixed`` gegenüber ``vlsm`` (Satz "VLSM dominiert").""" 

28 if not sizes: 

29 raise PlanError("Keine Blockgrößen angegeben") 

30 return len(sizes) * max(sizes) - sum(sizes)

33def plan_statistics(plan: Plan, cfg: PlanConfig) -> PlanStats: 

34 """Blockauslastung je Gruppe sowie Mittelwert, Minimum und Verschnitt.""" 

35 by_group = {g.name: g for g in cfg.groups} 

36 util: Dict[str, float] = {} 

37 required = 0 

38 allocated = 0 

39 for s in plan.subnets: 

40 need = required_addresses(by_group[s.group], cfg.growth_factor, cfg.gateway_position, s.network.version) 

41 required += need 

42 allocated += s.network.num_addresses 

43 util[s.group] = need / s.network.num_addresses 

44 values = list(util.values()) 

45 return PlanStats(util, sum(values) / len(values), min(values), allocated, required, allocated - required)

48def simulate_mean_utilization(samples: int = 100_000, seed: int = 1, decades: float = 6.0) -> float: 

49 """Monte-Carlo-Schätzung der mittleren Blockauslastung bei gleichverteiltem log2(r) mod 1.""" 

50 if samples < 1: 

51 raise PlanError("samples muss mindestens 1 sein") 

52 rng = random.Random(seed) 

53 total = 0.0 

54 for _ in range(samples): 

55 x = rng.uniform(1.0, 1.0 + decades * math.log2(10)) # Größen über mehrere Größenordnungen 

56 total += 2.0 ** (x - math.ceil(x)) 

57 return total / samples

60def poisson_cdf(k: int, lam: float) -> float: 

61 """P[H <= k] für H ~ Poisson(lam), numerisch stabil über Log-Terme.""" 

62 if lam < 0 or k < 0: 

63 raise PlanError("lam und k müssen nichtnegativ sein") 

64 if lam == 0: 

65 return 1.0 

66 total = 0.0 

67 for i in range(k + 1): 

68 total += math.exp(-lam + i * math.log(lam) - math.lgamma(i + 1)) 

69 return min(total, 1.0)

72def poisson_block_exponent(lam: float, overhead: int, alpha: float, max_exponent: int = 40) -> int: 

73 """Kleinstes k mit P[H + overhead <= 2^k] >= 1 - alpha für H ~ Poisson(lam). 

74 

75 Dimensionierung der Wachstumsreserve unter einem Wachstumsmodell (Ausblick). 

76 """ 

77 if not 0.0 < alpha < 1.0: 

78 raise PlanError("alpha muss echt zwischen 0 und 1 liegen") 

79 if overhead < 0: 

80 raise PlanError("overhead muss nichtnegativ sein") 

81 for k in range(max_exponent + 1): 

82 limit = (1 << k) - overhead 

83 if limit >= 0 and poisson_cdf(limit, lam) >= 1.0 - alpha: 

84 return k 

85 raise PlanError("Kein Blockexponent bis 2^%d erfüllt die Anforderung" % max_exponent)

88def sites_bound_holds(plan: Plan, cfg: PlanConfig) -> Tuple[bool, int, int, int]: 

89 """Prüft Satz "Zweistufige Planung": sum(E) < 2*sum(s) < 4*sum(r) (nur bei vlsm, ohne Überschreibungen).""" 

90 if not plan.site_blocks: 

91 raise PlanError("Der Plan hat keine Standortblöcke") 

92 total_e = sum(net.num_addresses for _, net in plan.site_blocks) 

93 total_s = sum(s.network.num_addresses for s in plan.subnets) 

94 total_r = sum( 

95 required_addresses(g, cfg.growth_factor, cfg.gateway_position, plan.base_network.version) for g in cfg.groups 

96 ) 

97 return total_e < 2 * total_s < 4 * total_r, total_e, total_s, total_r
